from mage_ai.data_preparation.decorators import data_exporter
from sqlalchemy import create_engine, URL
import pandas as pd


@data_exporter
def export_data(data, *args, **kwargs):
    # 앞 블록의 결과를 DataFrame으로 변환
    df = data if isinstance(data, pd.DataFrame) else pd.DataFrame(data)

    db_url = URL.create(
        drivername='postgresql+pg8000',
        username='postgres',
        password='rlaeodud1!',
        host='localhost',
        port=5432,
        database='postgres',
    )

    engine = create_engine(db_url)

    try:
        with engine.begin() as connection:
            df.to_sql(
                name='job_metadata',
                con=connection,
                schema='public',
                if_exists='append',
                index=False,
            )

        print(f'job_metadata 테이블에 {len(df)}건 저장 완료')

    finally:
        engine.dispose()
